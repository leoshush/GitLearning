select sroll, firstname ,dob ,cityname ,statename from student natural join connectable natural join statetable natural join citytable; 

/* alpha join */

select a.sroll,a.firstname,a.dob,b.cityname,c.statename from student a , connectable d, citytable b, statetable c where (a.sroll = d.sroll) and (b.cityid = d.cityid) and (b.stateid = c.stateid);

/* beta join */

select a.sroll , a.firstname , a.dob , b.cityname , c.statename from student a inner join connectable d on (a.sroll = d.sroll ) 
inner join citytable b on (b.cityid = d.cityid)
inner join statetable c on (c.stateid = b.stateid);

select a.cityname , count(b.sroll) from student b inner join  connectable d on (b.sroll = d.sroll) inner join citytable a group by a.cityid;

 select count(a.cityid) as TotalCities , b.statename from citytable a , statetable b WHERE a.stateid = b.stateid GROUP BY a.stateid , b.stateid;
