-- List the students whose test performance is greater than 50 % in the first test --

select sroll , firstname from student 
where sroll in 
(select sroll from markstable where testnum = 1 and marks > 50); 

-- average performance should be greater than 50 --

select sroll , firstname from student where sroll in (select sroll from markstable where marks > 70  in  (select avg(marks) from markstable group by sroll));

select sroll , firstname from student where sroll in (select sroll from markstable group by sroll having avg(marks)>70);

-- list student id ,student name ,test num and average performance testwise --

select a.sroll , a.firstname , b.testnum , b.ay from student a inner join (select sroll , avg(marks) as ay ,testnum from markstable group by sroll , testnum) b on (a.sroll = b.sroll) ; 

-- list the student with the marks with average performance for first test with the result fail pass , first class distinction --
 -- inline view--
	(select sroll , avg(marks) av ,testnum from markstable group by sroll ,testnum )  ;
    
select a.sroll , a.firstname , b.av ,
case 
     when b.av<35 then 'Fail'
	 when b.av>35 then 'Pass'
     when b.av>50 then 'First Class'
     when b.av>85 then 'Distinction'
end as result     
from student a
inner join (select sroll , avg(marks) av from markstable where testnum = 1 group by sroll) b on (a.sroll = b.sroll);
